# PANDUAN EKSEKUSI TAHAP 6 & 7: INTEGRASI MULTI-TAHUN DATA SATELIT KE WEBGIS

import React, { useEffect, useState } from 'react';
import { MapContainer, TileLayer, GeoJSON, LayersControl } from 'react-leaflet';
import 'leaflet/dist/leaflet.css';
import './App.css';

const { BaseLayer, Overlay } = LayersControl;

function App() {
  const centerJabar = [-6.9175, 107.6191];
  const zoomLevel = 8;
  const [geoData, setGeoData] = useState(null);

  // State pilihan tahun (default: 2023)
  const [selectedYear, setSelectedYear] = useState('2023');

  // =========================================================================
  // 1. DATA STATIS: KEMIRINGAN LERENG (SLOPE)
  // =========================================================================
  const TILE_SLOPE_URL = "[https://earthengine.googleapis.com/v1/projects/tugas-tpbw-510613/maps/2576756ac61810f94f5774bc50b10d8a-5c98fa99ca37c98b8bf0ebc267f34980/tiles/](https://earthengine.googleapis.com/v1/projects/tugas-tpbw-510613/maps/2576756ac61810f94f5774bc50b10d8a-5c98fa99ca37c98b8bf0ebc267f34980/tiles/){z}/{x}/{y}";

  // =========================================================================
  // 2. DATA DINAMIS: CITRA OPTIK SENTINEL-2 RGB (2021 - 2025)
  // =========================================================================
  const S2_TILES = {
    '2021': "[https://earthengine.googleapis.com/v1/projects/tugas-tpbw-510613/maps/e526a541aee7188a6b7d8d4bc93f37f0-52daf0b6b6b9f4f9cf67d1df459e88b5/tiles/](https://earthengine.googleapis.com/v1/projects/tugas-tpbw-510613/maps/e526a541aee7188a6b7d8d4bc93f37f0-52daf0b6b6b9f4f9cf67d1df459e88b5/tiles/){z}/{x}/{y}",
    '2022': "[https://earthengine.googleapis.com/v1/projects/tugas-tpbw-510613/maps/4ca29d8315b1d76c869330e88ee6bac0-4364a03d10434eb4be231a761c1d839f/tiles/](https://earthengine.googleapis.com/v1/projects/tugas-tpbw-510613/maps/4ca29d8315b1d76c869330e88ee6bac0-4364a03d10434eb4be231a761c1d839f/tiles/){z}/{x}/{y}",
    '2023': "[https://earthengine.googleapis.com/v1/projects/tugas-tpbw-510613/maps/2576756ac61810f94f5774bc50b10d8a-4c2a58aadcc2649ab1983c8ce7ca0099/tiles/](https://earthengine.googleapis.com/v1/projects/tugas-tpbw-510613/maps/2576756ac61810f94f5774bc50b10d8a-4c2a58aadcc2649ab1983c8ce7ca0099/tiles/){z}/{x}/{y}",
    '2024': "[https://earthengine.googleapis.com/v1/projects/tugas-tpbw-510613/maps/d8b45a8c414b5afb754284972bdaed80-a4ff686438f1528622496aa7bc9c0048/tiles/](https://earthengine.googleapis.com/v1/projects/tugas-tpbw-510613/maps/d8b45a8c414b5afb754284972bdaed80-a4ff686438f1528622496aa7bc9c0048/tiles/){z}/{x}/{y}",
    '2025': "[https://earthengine.googleapis.com/v1/projects/tugas-tpbw-510613/maps/c9712366c6ad85cfac3b0296a1ca95cc-a774a6634686e461b0903c97483247f6/tiles/](https://earthengine.googleapis.com/v1/projects/tugas-tpbw-510613/maps/c9712366c6ad85cfac3b0296a1ca95cc-a774a6634686e461b0903c97483247f6/tiles/){z}/{x}/{y}",
  };

  // =========================================================================
  // 3. DATA DINAMIS: CURAH HUJAN CHIRPS (2021 - 2025)
  // =========================================================================
  const CHIRPS_TILES = {
    '2021': "[https://earthengine.googleapis.com/v1/projects/tugas-tpbw-510613/maps/83e130adab5b267d09c9693dc6179414-de6510cb6bbe76aea699782b8b0b3a57/tiles/](https://earthengine.googleapis.com/v1/projects/tugas-tpbw-510613/maps/83e130adab5b267d09c9693dc6179414-de6510cb6bbe76aea699782b8b0b3a57/tiles/){z}/{x}/{y}",

  };

  // Membaca file GeoJSON Batas Administrasi
  useEffect(() => {
    fetch('/data/jabar-adm.geojson')
      .then((res) => {
        if (!res.ok) throw new Error('Gagal mengambil file GeoJSON');
        return res.json();
      })
      .then((data) => setGeoData(data))
      .catch((err) => console.error('Error GeoJSON:', err));
  }, []);

  const geoJsonStyle = {
    color: '#1d4ed8',
    weight: 1.5,
    opacity: 0.9,
    fillColor: '#3b82f6',
    fillOpacity: 0.05
  };

  const onEachFeature = (feature, layer) => {
    const name = feature.properties?.shapeName || feature.properties?.NAME_2 || 'Wilayah Jawa Barat';
    layer.bindPopup(`<strong>${name}</strong>`);
  };

  return (
    <div className="map-container" style={{ position: 'relative', width: '100vw', height: '100vh' }}>
      
      {/* Panel Pemilih Tahun di Kiri Atas */}

        <LayersControl position="topright">
          {/* Peta Dasar OpenStreetMap */}
          <BaseLayer checked name="OpenStreetMap">
            <TileLayer attribution="&copy; <a href=" [https://www.openstreetmap.org/copyright](https://www.openstreetmap.org/copyright)"">OpenStreetMap</a> contributors'
              url="https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png"
            />
          </BaseLayer>

          {/* Batas Administrasi */}
          {geoData && (
            <Overlay checked name="Batas Administrasi Jabar">
              <GeoJSON data="{geoData}" onEachFeature="{onEachFeature}" style="{geoJsonStyle}"/>
            </Overlay>
          )}

          {/* Kemiringan Lereng (Statis) */}
          <Overlay checked name="Kemiringan Lereng (Slope)">
            <TileLayer attribution="USGS SRTM via Google Earth Engine" opacity="{0.65}" url="{TILE_SLOPE_URL}"/>
          </Overlay>

          {/* Sentinel-2 Dinamis Sesuai Tahun Terpilih */}
          <Overlay (${selectedYear})`} RGB key="{`s2-${selectedYear}`}" name="{`Sentinel-2">
            <TileLayer attribution="Copernicus Sentinel-2 via Google Earth Engine" opacity="{0.8}" url="{S2_TILES[selectedYear]}"/>
          </Overlay>

          {/* Curah Hujan CHIRPS Dinamis Sesuai Tahun Terpilih */}
          <Overlay (${selectedYear})`} CHIRPS Hujan checked key="{`chirps-${selectedYear}`}" name="{`Curah">
            <TileLayer attribution="UCSB CHIRPS via Google Earth Engine" opacity="{0.65}" url="{CHIRPS_TILES[selectedYear]}"/>
          </Overlay>
        </LayersControl>
      </MapContainer>
    </div>
  );
}

export default App;
